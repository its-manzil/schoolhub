"use client";

import * as React from 'react';
import { useState, useEffect, useCallback } from 'react';
import { Eye, EyeOff } from "lucide-react";
// import './profile.css';
import SchoolAdminNavbar from '../../Components/SchoolAdminNavbar';
import BillingNavbar from '@/app/billing/components/BillingNavbar';
interface BillingAccount {
  _id: string;
  username: string;
  role: string;
  schoolId: number;
  isActive: boolean;
  createdAt: string;
  updatedAt?: string;
  lastLogin?: string | null;
}

interface UserData {
  schoolId?: number;
  [key: string]: unknown;
}

const BillingProfilePage = () => {
  const [billingAccount, setBillingAccount] = useState<BillingAccount | null>(null);
  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null);

  // Profile Settings State
  const [generatedUsername, setGeneratedUsername] = useState('');
  const [createPassword, setCreatePassword] = useState('');
  const [createConfirmPassword, setCreateConfirmPassword] = useState('');
  const [showCreatePassword, setShowCreatePassword] = useState(false);
  const [showCreateConfirmPassword, setShowCreateConfirmPassword] = useState(false);
  const [createLoading, setCreateLoading] = useState(false);
  const [deleteLoading, setDeleteLoading] = useState(false);
  const [deletePassword, setDeletePassword] = useState('');
  const [showDeleteModal, setShowDeleteModal] = useState(false);

  // Password Change State (for existing billing account)
  const [newPassword, setNewPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');
  const [showNewPassword, setShowNewPassword] = useState(false);
  const [showConfirmNewPassword, setShowConfirmNewPassword] = useState(false);
  const [passwordLoading, setPasswordLoading] = useState(false);

  const showMessage = (type: 'success' | 'error', text: string) => {
    setMessage({ type, text });
    setTimeout(() => setMessage(null), 4000);
  };

  // Safely parse JSON from localStorage (for objects like 'user')
  const safeJSONParse = <T,>(key: string): T | null => {
    if (typeof window === 'undefined') return null;
    try {
      const item = localStorage.getItem(key);
      if (!item) return null;
      return JSON.parse(item) as T;
    } catch {
      return null;
    }
  };

  // Get token directly (stored as plain string, not JSON)
  const getToken = (): string | null => {
    if (typeof window === 'undefined') return null;
    return localStorage.getItem('token');
  };

  const fetchBillingAccount = useCallback(async () => {
    try {
      const token = getToken();
      if (!token) {
        showMessage('error', 'You are not logged in');
        return;
      }

      const user = safeJSONParse<UserData>('user');
      if (!user || !user.schoolId) {
        showMessage('error', 'School ID not found in user credentials');
        return;
      }

      // Generate username from schoolId in correct format: bill_{schoolId}
      const username = `bill_${user.schoolId}`;
      setGeneratedUsername(username);

      // Fetch billing account data
      const response = await fetch('http://localhost:2056/api/users/billing', {
        headers: {
          Authorization: `Bearer ${token}`
        }
      });

      const data = await response.json();

      if (!response.ok || !data.success) {
        // No billing account exists yet - this is expected behavior
        setBillingAccount(null);
        return;
      }

      setBillingAccount(data.data);
    } catch (error) {
      console.error('Error loading billing account:', error);
      showMessage('error', 'Failed to load billing account');
    }
  }, []);

  useEffect(() => {
    fetchBillingAccount();
  }, [fetchBillingAccount]);

  const handleCreateBillingAccount = async () => {
    try {
      if (!createPassword || !createConfirmPassword) {
        showMessage('error', 'Password fields are required');
        return;
      }

      if (createPassword !== createConfirmPassword) {
        showMessage('error', 'Password and confirm password do not match');
        return;
      }

      if (createPassword.length < 6) {
        showMessage('error', 'Password must be at least 6 characters');
        return;
      }

      const token = getToken();
      if (!token) {
        showMessage('error', 'You are not logged in');
        return;
      }

      const user = safeJSONParse<UserData>('user');
      if (!user || !user.schoolId) {
        showMessage('error', 'School ID not found');
        return;
      }

      setCreateLoading(true);

      const response = await fetch('http://localhost:2056/api/users/create-billing', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${token}`
        },
        body: JSON.stringify({
          username: `bill_${user.schoolId}`,
          password: createPassword,
          role: 'billing',
          schoolId: user.schoolId
        })
      });

      const data = await response.json();

      if (!response.ok || !data.success) {
        showMessage('error', data.message || 'Failed to create billing account');
        return;
      }

      setBillingAccount(data.data);
      setCreatePassword('');
      setCreateConfirmPassword('');
      showMessage('success', 'Billing account created successfully');
    } catch (error) {
      console.error('Error creating billing account:', error);
      showMessage('error', 'Failed to create billing account');
    } finally {
      setCreateLoading(false);
    }
  };

  const handleDeleteBillingAccount = async () => {
    try {
      if (!deletePassword) {
        showMessage('error', 'School admin password is required to delete billing account');
        return;
      }

      const token = getToken();
      if (!token) {
        showMessage('error', 'You are not logged in');
        return;
      }

      setDeleteLoading(true);

      const response = await fetch('http://localhost:2056/api/users/delete-billing', {
        method: 'DELETE',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${token}`
        },
        body: JSON.stringify({
          adminPassword: deletePassword
        })
      });

      const data = await response.json();

      if (!response.ok || !data.success) {
        showMessage('error', data.message || 'Failed to delete billing account');
        return;
      }

      setBillingAccount(null);
      setDeletePassword('');
      setShowDeleteModal(false);
      showMessage('success', 'Billing account deleted successfully');
    } catch (error) {
      console.error('Error deleting billing account:', error);
      showMessage('error', 'Failed to delete billing account');
    } finally {
      setDeleteLoading(false);
    }
  };

  const handleChangeBillingPassword = async () => {
    try {
      if (!newPassword || !confirmPassword) {
        showMessage('error', 'Please fill both password fields');
        return;
      }

      if (newPassword !== confirmPassword) {
        showMessage('error', 'Passwords do not match');
        return;
      }

      if (newPassword.length < 6) {
        showMessage('error', 'Password must be at least 6 characters');
        return;
      }

      const token = getToken();
      if (!token) {
        showMessage('error', 'You are not logged in');
        return;
      }

      setPasswordLoading(true);

      const response = await fetch('http://localhost:2056/api/users/change-billing-password', {
        method: 'PUT',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${token}`
        },
        body: JSON.stringify({
          newPassword
        })
      });

      const data = await response.json();

      if (!response.ok || !data.success) {
        showMessage('error', data.message || 'Failed to update billing password');
        return;
      }

      setNewPassword('');
      setConfirmPassword('');
      showMessage('success', 'Billing password updated successfully');
    } catch (error) {
      console.error('Error updating billing password:', error);
      showMessage('error', 'Failed to update billing password');
    } finally {
      setPasswordLoading(false);
    }
  };

  return (
    <>
      <SchoolAdminNavbar />
      <BillingNavbar />
    <div className="billing-settings-layout">
      <div className="billing-settings-sidebar">
        <h2 className="billing-settings-title">Billing Profile</h2>
      </div>

      <div className="billing-settings-main">
        <div className="billing-settings-content">
          {message && (
            <div className={`billing-message ${message.type === 'success' ? 'billing-message-success' : 'billing-message-error'}`}>
              {message.text}
            </div>
          )}

          {billingAccount ? (
            <div className="billing-chart-card">
              <h2 className="billing-chart-title">Billing Account Details</h2>
              <div className="billing-account-info">
                <div className="billing-account-detail">
                  <strong>Username:</strong> {billingAccount.username}
                </div>
                <div className="billing-account-detail">
                  <strong>Role:</strong> {billingAccount.role}
                </div>
                <div className="billing-account-detail">
                  <strong>Status:</strong>
                  <span className={`billing-status ${billingAccount.isActive ? 'billing-status-active' : 'billing-status-inactive'}`}>
                    {billingAccount.isActive ? 'Active' : 'Inactive'}
                  </span>
                </div>
                <div className="billing-account-detail">
                  <strong>Created:</strong> {new Date(billingAccount.createdAt).toLocaleDateString()}
                </div>
                {billingAccount.lastLogin && (
                  <div className="billing-account-detail">
                    <strong>Last Login:</strong> {new Date(billingAccount.lastLogin).toLocaleDateString()}
                  </div>
                )}
              </div>

              <div className="billing-account-actions">
                <h3>Change Billing Password</h3>
                <div className="billing-form">
                  <div className="billing-form-row">
                    <label>
                      New Password
                      <div className="billing-password-input-wrapper">
                        <input
                          type={showNewPassword ? "text" : "password"}
                          value={newPassword}
                          onChange={e => setNewPassword(e.target.value)}
                          className="billing-input"
                          placeholder="Enter new password"
                        />
                        <button
                          type="button"
                          className="billing-password-toggle"
                          onClick={() => setShowNewPassword(!showNewPassword)}
                        >
                          {showNewPassword ? <EyeOff size={20} /> : <Eye size={20} />}
                        </button>
                      </div>
                    </label>
                    <label>
                      Confirm New Password
                      <div className="billing-password-input-wrapper">
                        <input
                          type={showConfirmNewPassword ? "text" : "password"}
                          value={confirmPassword}
                          onChange={e => setConfirmPassword(e.target.value)}
                          className="billing-input"
                          placeholder="Confirm new password"
                        />
                        <button
                          type="button"
                          className="billing-password-toggle"
                          onClick={() => setShowConfirmNewPassword(!showConfirmNewPassword)}
                        >
                          {showConfirmNewPassword ? <EyeOff size={20} /> : <Eye size={20} />}
                        </button>
                      </div>
                    </label>
                  </div>
                  <div className="billing-form-actions">
                    <button
                      className="billing-btn primary"
                      onClick={handleChangeBillingPassword}
                      disabled={passwordLoading}
                    >
                      {passwordLoading ? 'Updating...' : 'Change Password'}
                    </button>
                    <button
                      className="billing-btn danger"
                      onClick={() => setShowDeleteModal(true)}
                    >
                      Delete Account
                    </button>
                  </div>
                </div>
              </div>
            </div>
          ) : (
            <div className="billing-chart-card">
              <h2 className="billing-chart-title">Create Billing Account</h2>
              <p className="billing-td-muted">Create a new billing account for your school.</p>

              <div className="billing-form">
                <div className="billing-form-row">
                  <div className="billing-username-display">
                    <strong>Username:</strong> {generatedUsername}
                  </div>
                </div>
                <div className="billing-form-row">
                  <label>
                    Password
                    <div className="billing-password-input-wrapper">
                      <input
                        type={showCreatePassword ? "text" : "password"}
                        value={createPassword}
                        onChange={e => setCreatePassword(e.target.value)}
                        className="billing-input"
                        placeholder="Enter password"
                      />
                      <button
                        type="button"
                        className="billing-password-toggle"
                        onClick={() => setShowCreatePassword(!showCreatePassword)}
                      >
                        {showCreatePassword ? <EyeOff size={20} /> : <Eye size={20} />}
                      </button>
                    </div>
                  </label>
                  <label>
                    Confirm Password
                    <div className="billing-password-input-wrapper">
                      <input
                        type={showCreateConfirmPassword ? "text" : "password"}
                        value={createConfirmPassword}
                        onChange={e => setCreateConfirmPassword(e.target.value)}
                        className="billing-input"
                        placeholder="Confirm password"
                      />
                      <button
                        type="button"
                        className="billing-password-toggle"
                        onClick={() => setShowCreateConfirmPassword(!showCreateConfirmPassword)}
                      >
                        {showCreateConfirmPassword ? <EyeOff size={20} /> : <Eye size={20} />}
                      </button>
                    </div>
                  </label>
                </div>
                <div className="billing-form-actions">
                  <button
                    className="billing-btn primary"
                    onClick={handleCreateBillingAccount}
                    disabled={createLoading}
                  >
                    {createLoading ? 'Creating...' : 'Create Account'}
                  </button>
                </div>
              </div>
            </div>
          )}

          {/* Delete Account Modal */}
          {showDeleteModal && (
            <div className="billing-modal-overlay">
              <div className="billing-modal">
                <h3>Delete Billing Account</h3>
                <p>This action cannot be undone. Please enter your school admin password to confirm.</p>

                <div className="billing-form">
                  <label>
                    School Admin Password
                    <div className="billing-password-input-wrapper">
                      <input
                        type="password"
                        value={deletePassword}
                        onChange={e => setDeletePassword(e.target.value)}
                        className="billing-input"
                        placeholder="Enter your admin password"
                      />
                    </div>
                  </label>

                  <div className="billing-form-actions">
                    <button
                      className="billing-btn danger"
                      onClick={handleDeleteBillingAccount}
                      disabled={deleteLoading}
                    >
                      {deleteLoading ? 'Deleting...' : 'Delete Account'}
                    </button>
                    <button
                      className="billing-btn-outline"
                      onClick={() => {
                        setShowDeleteModal(false);
                        setDeletePassword('');
                      }}
                    >
                      Cancel
                    </button>
                  </div>
                </div>
              </div>
            </div>
          )}
        </div>
      </div>
    </div>
    </>
  );
};

export default BillingProfilePage;
